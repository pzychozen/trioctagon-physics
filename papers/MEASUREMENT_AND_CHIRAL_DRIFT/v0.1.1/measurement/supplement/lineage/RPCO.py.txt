# rpco_field.py
import numpy as np
import matplotlib.pyplot as plt

def rpc_stabilizer_field(grid_size=100, decay=0.05, frequency=5):
    x = np.linspace(-2*np.pi, 2*np.pi, grid_size)
    y = np.linspace(-2*np.pi, 2*np.pi, grid_size)
    X, Y = np.meshgrid(x, y)

    R = np.sqrt(X**2 + Y**2)
    Theta = np.arctan2(Y, X)
    field = np.sin(frequency * Theta) * np.exp(-decay * R**2)
    return X, Y, field

X, Y, rpc_field = rpc_stabilizer_field()

plt.figure(figsize=(8, 6))
plt.contourf(X, Y, rpc_field, levels=100, cmap='plasma')
plt.title("RPCO – Recursive Phase Collapse Operator")
plt.xlabel("X")
plt.ylabel("Y")
plt.colorbar(label="Field Intensity")
plt.axis('equal')
plt.tight_layout()
plt.show()

# Save to txt
np.savetxt("rpco_field_values.txt", np.column_stack((X.flatten(), Y.flatten(), rpc_field.flatten())), header="X Y Field", fmt="%.5f")
