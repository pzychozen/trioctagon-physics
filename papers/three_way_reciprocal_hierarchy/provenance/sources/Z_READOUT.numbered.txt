0001: """Opt-in Paper-E readouts; never a recurrence, geometry map or feedback law.
0002: 
0003: Binary64/complex128 evaluation reuses the conservative response precision
0004: contract. Intermediate nonzero subnormals, overflow and lost nonzero products
0005: raise ResponsePrecisionError. Successful scalar evaluation does not guarantee
0006: that cubic, chirality or blended-vector evaluation will succeed.
0007: """
0008: from dataclasses import dataclass, field
0009: import math
0010: import numbers
0011: 
0012: import numpy as np
0013: 
0014: from . import readouts
0015: from ._response_numeric import (
0016:     ResponsePrecisionError, checked_real, complex_product, complex_vector,
0017:     product, real_scalar, total,
0018: )
0019: 
0020: 
0021: def _integer(value, name):
0022:     if isinstance(value, (bool, np.bool_)) or not isinstance(value, numbers.Integral):
0023:         raise TypeError(f"{name} must be an integer, excluding booleans")
0024:     return int(value)
0025: 
0026: 
0027: def _require(value, cls, name):
0028:     if not isinstance(value, cls):
0029:         raise TypeError(f"{name} must be {cls.__name__}")
0030:     return value
0031: 
0032: 
0033: def _freeze(values):
0034:     result = np.array(values, copy=True)
0035:     result.setflags(write=False)
0036:     return result
0037: 
0038: 
0039: def _real_vector(values, name):
0040:     raw = np.asarray(values, dtype=object)
0041:     if raw.shape != (3,):
0042:         raise ValueError(f"{name} must have shape (3,)")
0043:     return _freeze([checked_real(real_scalar(v, name), name) for v in raw])

0046: @dataclass(frozen=True, slots=True)
0047: class Clock:
0048:     """Explicit clock; q stays integer. Historical N=12, q_step=1.
0049: 
0050:     q is retained as supplied; angle evaluation first reduces q modulo N
0051:     using integer arithmetic. This periodic evaluation never changes Omega.
0052:     """
0053:     q: int = 0
0054:     N: int = 12
0055:     t: float = 0.0
0056:     q_step: int = 1
0057: 
0058:     def __post_init__(self):
0059:         for name in ("q", "N", "q_step"):
0060:             object.__setattr__(self, name, _integer(getattr(self, name), name))
0061:         if self.N <= 0:
0062:             raise ValueError("N must be positive")
0063:         object.__setattr__(self, "t", real_scalar(self.t, "t"))
0064: 
0065: 
0066: def clock_angle(clock):
0067:     """2*pi*(q mod N)/N; no conversion of a huge raw q to binary64."""
0068:     _require(clock, Clock, "clock")
0069:     q = clock.q % clock.N
0070:     ratio = q / clock.N
0071:     ratio = checked_real(ratio, "clock sector fraction", nonzero=(q != 0))
0072:     return product(2.0 * math.pi, ratio, "clock angle")
0073: 
0074: 
0075: def advance_clock(clock, dt):
0076:     """Return a new clock; dt is explicit (historical runner used 0.1)."""
0077:     _require(clock, Clock, "clock")
0078:     dt = real_scalar(dt, "dt")
0079:     t = total((clock.t, dt), "clock time addition")
0080:     if dt != 0 and t == clock.t:
0081:         raise ResponsePrecisionError("nonzero dt lost: clock time did not advance")
0082:     return Clock((clock.q + clock.q_step) % clock.N, clock.N, t, clock.q_step)

0085: def _config_reals(config, names):
0086:     for name in names:
0087:         object.__setattr__(config, name, real_scalar(getattr(config, name), name))

0091: class StagedConfig:
0092:     """Literal K defaults; signed finite coefficients are allowed."""

0095:     theta_lock: float = 0.244
0096:     alpha: float = 1.0

0103: def state_norm(omega):
0104:     """Stable six-real-component hypot, retaining raw state scale."""
0105:     v = complex_vector(omega, 3, "omega")
0106:     return checked_real(math.hypot(*(float(c) for z in v for c in (z.real, z.imag))),
0107:                         "state norm", nonzero=bool(np.any(v != 0)))
0108: 
0109: 
0110: def saturated_norm(omega):
0111:     """rho=kappa/(1+kappa); floating rho may round to 1."""
0112:     kappa = state_norm(omega)
0113:     denominator = total((1.0, kappa), "rho denominator")
0114:     return checked_real(kappa / denominator, "rho", nonzero=(kappa != 0))
0115: 
0116: 
0117: def _harmonic(omega, clock, config):
0118:     rho = saturated_norm(omega)
0119:     theta = clock_angle(clock)
0120:     angle = product(3.0, total((theta, -config.theta_lock), "lock difference"),
0121:                     "harmonic angle")
0122:     cosine = checked_real(math.cos(angle), "harmonic cosine")
0123:     return product(product(config.lambda_vp, rho, "harmonic amplitude"),
0124:                    cosine, "harmonic scalar")
0125: 
0126: 
0127: def staged_scalar(omega, clock, config):
0128:     """K scalar, left-associated checked products and an explicit envelope.
0129: 
0130:     No envelope bypass is made for zero amplitude; its numerical evaluation
0131:     still has to succeed. No finite-time underflow is labelled exact decay.
0132:     """
0133:     _require(config, StagedConfig, "config")
0134:     _require(clock, Clock, "clock")
0135:     harmonic = _harmonic(omega, clock, config)
0136:     exponent = product(-config.gamma, clock.t, "envelope exponent")
0137:     try:
0138:         envelope = math.exp(exponent)
0139:     except OverflowError as exc:
0140:         raise ResponsePrecisionError("exponential envelope overflow") from exc
0141:     envelope = checked_real(envelope, "exponential envelope", nonzero=True)
0142:     return product(harmonic, envelope, "staged scalar")

0145: def macro_vector(z, clock):
0146:     """Fresh read-only M=z*(cos(theta),sin(theta),1), without normalization."""
0147:     z = checked_real(real_scalar(z, "z"), "z")
0148:     theta = clock_angle(clock)
0149:     return _freeze([product(z, math.cos(theta), "macro x"),
0150:                     product(z, math.sin(theta), "macro y"), z])

0152: 
0153: def blend_vectors(macro, chiral, *, alpha, beta):
0154:     """Fresh read-only alpha*M+beta*C; validate both vectors even at zero weight."""
0155:     alpha = real_scalar(alpha, "alpha")
0156:     beta = real_scalar(beta, "beta")
0157:     m = _real_vector(macro, "macro")
0158:     c = _real_vector(chiral, "chiral")
0159:     return _freeze([total((product(alpha, x, "macro blend"),
0160:                            product(beta, y, "chiral blend")), "total blend")
0161:                     for x, y in zip(m, c)])

0226: def cubic_j(omega):
0227:     """J=Im(O1*conj(O2)*O3), with two left-associated checked complex products."""
0228:     v = complex_vector(omega, 3, "omega")
0229:     pair = complex_product(complex(v[0]), complex(v[1]).conjugate(), "cubic pair")
0230:     return complex_product(pair, complex(v[2]), "cubic triple").imag

0235:     j = cubic_j(omega)
0236:     denominator = total((1.0, abs(j)), "cubic normalization denominator")
0237:     return checked_real(j / denominator, "normalized cubic", nonzero=(j != 0))

0240: def advance_ema(omega, memory):
0241:     """Exactly one pure H update using the newly committed Omega.
0242: 
0243:     The source innovation is literal 0.01, not 1-float(0.99).
0244:     No clock or recurrence update is performed.

0246:     _require(memory, EMAState, "memory")
