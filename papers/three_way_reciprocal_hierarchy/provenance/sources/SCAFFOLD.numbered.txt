0001: """Opt-in exact Paper-D v0.1.1 reference geometry, equations (1)-(29).
0002: 
0003: Lengths accept Python int/Fraction and exact SymPy real finite expressions.
0004: Floats (including nested SymPy Float), strings and booleans are rejected.
0005: Positivity must be provable: use positive symbols for s and g_gap. Unknown
0006: regularity and point membership return None, never a certified False/True.
0007: No numerical tolerance, geometry registry, welding, state or dynamics is used.
0008: """
0009: 
0010: from dataclasses import dataclass
0011: from fractions import Fraction
0012: 
0013: import sympy as sp
0014: 
0015: Point = tuple[sp.Expr, ...]
0016: Segment = tuple[Point, Point]
0017: _HALF = sp.Rational(1, 2)
0018: _R3 = sp.sqrt(3)
0019: RADIAL = ((sp.S.One, sp.S.Zero), (-_HALF, _R3/2), (-_HALF, -_R3/2))
0020: TANGENT = tuple((-y, x) for x, y in RADIAL)
0021: PAPER_C_SHRINK_FACTOR = (1 + sp.sqrt(2))/3
0022: 
0023: 
0024: def _real(value, name):
0025:     if isinstance(value, bool) or not isinstance(value, (int, Fraction, sp.Expr)):
0026:         raise TypeError(f"{name} requires an exact integer, Fraction or SymPy expression")
0027:     value = sp.sympify(value)
0028:     if value.has(sp.Float):
0029:         raise TypeError(f"{name} must be exact; floats are not rationalized")
0030:     if value.is_real is not True or value.is_finite is not True:
0031:         raise ValueError(f"{name} must be provably real and finite")
0032:     return sp.simplify(value)
0033: 
0034: 
0035: def _positive(value, name):
0036:     value = _real(value, name)
0037:     if value.is_positive is not True:
0038:         raise ValueError(f"{name} must be provably positive; undecidable domains are rejected")
0039:     return value
0040: 
0041: 
0042: def _point(*values):
0043:     # Rationalizing symbolic denominators can introduce removable poles at
0044:     # admitted inputs (e.g. 1/(1+sqrt(x)) at positive x=1). Keep those
0045:     # denominators intact; numerical radical denominators remain exact.
0046:     return tuple(sp.expand(sp.radsimp(sp.simplify(v), symbolic=False)) for v in values)
0047: 
0048: 
0049: def _input_point(values, dimension):
0050:     values = tuple(values)
0051:     if len(values) != dimension:
0052:         raise ValueError(f"point must have {dimension} coordinates")
0053:     return tuple(_real(v, "coordinate") for v in values)
0054: 
0055: 
0056: def _index(i):
0057:     if isinstance(i, bool) or not isinstance(i, (int, sp.Integer)):
0058:         raise TypeError("frame index must be an integer 0, 1 or 2")
0059:     if not 0 <= i < 3:
0060:         raise IndexError("frame index must be 0, 1 or 2")
0061:     return int(i)
0062: 
0063: 
0064: def _edges(vertices):
0065:     return tuple(zip(vertices, vertices[1:] + vertices[:1]))
0066: 
0067: 
0068: def _r60(v):
0069:     x, y = v
0070:     return _point(x/2 - _R3*y/2, _R3*x/2 + y/2)

0073: @dataclass(frozen=True, slots=True)
0074: class ConvexHull:
0075:     """Closed convex hull of exact generators, including its boundary.
0076: 
0077:     This is a set prescription, not material, a welded face, or a Boolean
0078:     polygon engine. Generated octagon/corner/triangle vertices are cyclic.
0079:     """
0080: 
0081:     vertices: tuple[Point, ...]
0082: 
0083:     def __post_init__(self):
0084:         vertices = tuple(tuple(p) for p in self.vertices)
0085:         if len(vertices) < 3 or len(vertices[0]) not in (2, 3):
0086:             raise ValueError("a reference hull needs at least three 2D or 3D generators")
0087:         object.__setattr__(self, "vertices", tuple(
0088:             _input_point(p, len(vertices[0])) for p in vertices))

0091: @dataclass(frozen=True, slots=True)
0092: class HalfPlane:
0093:     """The closed planar set normal dot x <= offset; normals need not be unit."""
0094: 
0095:     normal: tuple[sp.Expr, sp.Expr]
0096:     offset: sp.Expr
0097: 
0098:     def __post_init__(self):
0099:         normal = _input_point(self.normal, 2)
0100:         if sp.simplify(sum(x*x for x in normal)).is_positive is not True:
0101:             raise ValueError("half-plane normal must be provably nonzero")
0102:         object.__setattr__(self, "normal", normal)
0103:         object.__setattr__(self, "offset", _real(self.offset, "offset"))
0104: 
0105:     def slack(self, point):
0106:         """offset - normal dot point: nonnegative means retained, including zero."""
0107:         point = _input_point(point, 2)
0108:         return sp.simplify(self.offset - sum(x*y for x, y in zip(self.normal, point)))
